{{- if or .Values.ingress.enabled .Values.httpRoute.enabled }}
{{- else if eq .Values.guacamole.service.type "LoadBalancer" }}

{{- else }}
  kubectl port-forward -n {{ .Release.Namespace }} svc/{{ $svc }} 8080:{{ .Values.guacamole.service.port }}
  echo "http://127.0.0.1:8080{{ $path }}"
{{- end }}
